// 防止 XXE (XML External Entity) 攻击

use quick_xml::Reader;

/// XML 响应最大允许字节数 (10MB)
pub const MAX_XML_SIZE: usize = 10 * 1024 * 1024;

/// 创建安全配置的 XML Reader
/// - 禁用 DTD 解析（quick_xml 默认不解析 DTD）
/// - 启用 trim_text 减少内存占用
/// - 调用方应先检查输入大小不超过 MAX_XML_SIZE
pub fn create_safe_reader(xml: &str) -> Reader<&[u8]> {
    let mut reader = Reader::from_str(xml);
    reader.config_mut().trim_text(true);
    reader
}

/// 检查 XML 响应大小是否在安全范围内
pub fn validate_xml_size(xml: &str) -> Result<(), String> {
    if xml.len() > MAX_XML_SIZE {
        Err(format!("XML 响应过大: {} bytes (最大 {} bytes)", xml.len(), MAX_XML_SIZE))
    } else {
        Ok(())
    }
}
```

use aes_gcm::{aead::{Aead, KeyInit}, Aes256Gcm, Nonce};
use base64::{engine::general_purpose::STANDARD as BASE64, Engine};
use rand::Rng;
use hmac::{Hmac, Mac};
use pbkdf2::pbkdf2_hmac;
use sha2::Sha256;

type HmacSha256 = Hmac<Sha256>;

/// PBKDF2 迭代次数
const PBKDF2_ITERATIONS: u32 = 100_000;

/// 应用标识盐后缀
const SALT_SUFFIX: &[u8] = b"SmartRiverCalendar";

/// 生成随机盐值
pub fn generate_salt() -> [u8; 16] {
    let mut salt = [0u8; 16];
    rand::thread_rng().fill(&mut salt);
    salt
}

/// 基于用户 ID + 随机盐派生密钥（新方案）
pub fn derive_key_with_salt(user_id: i64, salt: &[u8]) -> [u8; 32] {
    let mut key = [0u8; 32];
    let password = user_id.to_le_bytes();
    let full_salt = [salt, SALT_SUFFIX].concat();
    pbkdf2_hmac::<Sha256>(&password, &full_salt, PBKDF2_ITERATIONS, &mut key);
    key
}

/// 旧方案密钥派生（用于数据迁移）
fn derive_key_legacy() -> [u8; 32] {
    let hostname = hostname::get()
        .unwrap_or_default()
        .to_string_lossy()
        .to_string();
    let salt = b"SmartRiverCalendar2024";
    let mut hasher = Sha256::new();
    hasher.update(hostname.as_bytes());
    hasher.update(salt);
    let result = hasher.finalize();
    let mut key = [0u8; 32];
    key.copy_from_slice(&result);
    key
}

/// 使用旧密钥解密数据
fn decrypt_with_key(key: &[u8; 32], encrypted_b64: &str) -> Result<String, String> {
    // 保留原有 decrypt 函数的核心逻辑，接受外部 key 参数
    let bytes = BASE64.decode(encrypted_b64).map_err(|e| e.to_string())?;
    if bytes.len() < 12 {
        return Err("密文过短".to_string());
    }
    let nonce = Nonce::from_slice(&bytes[..12]);
    let ciphertext = &bytes[12..];
    let cipher = Aes256Gcm::new_from_slice(key).map_err(|e| e.to_string())?;
    let plaintext = cipher.decrypt(nonce, ciphertext).map_err(|e| e.to_string())?;
    String::from_utf8(plaintext).map_err(|e| e.to_string())
}

/// 迁移加密数据：使用旧密钥解密后用新密钥重新加密
pub fn migrate_encrypted_data(
    user_id: i64,
    salt: &[u8],
    encrypted_b64: &str,
) -> Result<String, String> {
    let old_key = derive_key_legacy();
    let plaintext = decrypt_with_key(&old_key, encrypted_b64)?;
    let new_key = derive_key_with_salt(user_id, salt);
    encrypt_with_key(&new_key, &plaintext)
}

/// 使用指定密钥加密数据
fn encrypt_with_key(key: &[u8; 32], plaintext: &str) -> Result<String, String> {
    let cipher = Aes256Gcm::new_from_slice(key).map_err(|e| e.to_string())?;
    let nonce_bytes: [u8; 12] = rand::thread_rng().gen();
    let nonce = Nonce::from_slice(&nonce_bytes);
    let ciphertext = cipher.encrypt(nonce, plaintext.as_bytes()).map_err(|e| e.to_string())?;
    let mut result = nonce_bytes.to_vec();
    result.extend_from_slice(&ciphertext);
    Ok(BASE64.encode(&result))
}
```

注意：需保留原有的 `encrypt` 和 `decrypt` 公共函数签名，改为内部调用新方案。需要确认现有 `encrypt`/`decrypt` 的调用方签名。

// Web 端安全存储模块
// 使用 sessionStorage 替代 localStorage 存储 Token
// 关闭标签页后 Token 自动清除，降低 XSS 窃取风险

const ACCESS_TOKEN_KEY = 'src_access_token'
const REFRESH_TOKEN_KEY = 'src_refresh_token'

export function getStoredAccessToken(): string | null {
  try {
    return sessionStorage.getItem(ACCESS_TOKEN_KEY)
  } catch {
    return null
  }
}

export function getStoredRefreshToken(): string | null {
  try {
    return sessionStorage.getItem(REFRESH_TOKEN_KEY)
  } catch {
    return null
  }
}

export function storeTokens(access: string, refresh: string): void {
  try {
    sessionStorage.setItem(ACCESS_TOKEN_KEY, access)
    sessionStorage.setItem(REFRESH_TOKEN_KEY, refresh)
  } catch {
    console.warn('[SecureStorage] 无法存储 Token')
  }
}

export function clearStoredTokens(): void {
  try {
    sessionStorage.removeItem(ACCESS_TOKEN_KEY)
    sessionStorage.removeItem(REFRESH_TOKEN_KEY)
  } catch {
    // 忽略
  }
}

export function getRefreshToken(): string | null {
  return getStoredRefreshToken()
}
```

  try {
    const { authRepo } = usePlatform()
    const publicKeyBase64 = await authRepo.getPublicKey()

    const binaryString = atob(publicKeyBase64)
    const bytes = new Uint8Array(binaryString.length)
    for (let i = 0; i < binaryString.length; i++) {
      bytes[i] = binaryString.charCodeAt(i)
    }

    return await crypto.subtle.importKey(
      'spki',
      bytes.buffer,
      { name: 'RSA-OAEP', hash: 'SHA-256' },
      false,
      ['encrypt']
    )
  } catch (error) {
    console.error('[RSA] 获取公钥失败:', error)
    throw error
  }
}

export async function encryptPassword(password: string): Promise<string> {
  try {
    if (!cachedPublicKey) {
      cachedPublicKey = await fetchPublicKey()
    }

    const encrypted = await crypto.subtle.encrypt(
      { name: 'RSA-OAEP' },
      cachedPublicKey,
      new TextEncoder().encode(password)
    )

    const encryptedArray = new Uint8Array(encrypted)
    let binary = ''
    for (let i = 0; i < encryptedArray.length; i++) {
      binary += String.fromCharCode(encryptedArray[i])
    }
    return btoa(binary)
  } catch (error) {
    console.error('[RSA] 密码加密失败:', error)
    cachedPublicKey = null
    throw error
  }
}
```

    }

  const firstCalendar = calendars[0]

      return parsed
    }
  }
  console.warn('[CalendarHelpers] 无法获取有效的日历 ID，使用默认值 1')

}
